# The start screen: what it offers, and each way out of it.
# Run from the project folder (tests/ui/start_fixture.lsprite is a saved document): opening it puts
# one recent file on the screen to find.
open tests/ui/start_fixture.lsprite
wait 3
start
wait 3
expect item Create
expect item Open...
expect item Import sprite sheet...
expect item Library...
expect item Start with a blank canvas
expect item start_fixture.lsprite

# A tip is shown, and Next tip steps to another (it must not close the screen).
expect item Next tip
click-on Next tip
wait 2
expect item Next tip
expect item Create

# A recent file is already open here, so choosing it shows that tab.
click-on start_fixture.lsprite
wait 2
expect missing Create
expect tabs 1

# File > Start screen brings it back; leaving it is Start with a blank canvas.
click-on File
click-on Start screen
wait 2
expect item Create
click-on Start with a blank canvas
wait 2
expect missing Create

# A preset card fills in the details beside it and goes no further; Create
# makes the document. The one on screen has work in it (it was opened), so the
# new one gets a tab of its own.
click-on File
click-on Start screen
wait 2
expect item Icon 16 x 16
expect item Tile 32 x 32
expect item Character 64 x 64
expect item Portrait 128 x 128
expect item Scene 320 x 180
expect item Handheld 160 x 144
click-on Scene 320 x 180
wait 2
expect item Create
expect tabs 1
click-on Create
wait 2
expect missing Create
expect canvas 320 180
expect tabs 2

# Every detail can be set by hand: a preset, then a usual size, then a name.
# The untouched canvas from the card is replaced, so still two tabs.
click-on File
click-on Start screen
wait 2
click-on Icon 16 x 16
click-on 48
click-on ##newname
key ctrl+a
type hero walk
click-on Create
wait 2
expect canvas 48 48
expect item hero walk
expect tabs 2

# Swapping width and height turns a landscape preset to portrait.
click-on File
click-on Start screen
wait 2
click-on Handheld 160 x 144
click-on Swap width and height
click-on Create
wait 2
expect canvas 144 160

# File > New is the start screen: the same cards, the same answers as they were last set,
# and Ctrl+N is the same.
click-on File
click-on New...
wait 2
expect item Open...
expect item Library...
expect item Create
click-on Create
wait 2
expect canvas 144 160
key ctrl+n
wait 2
expect item Handheld 160 x 144
key escape
wait 2
expect missing Create

# A double click on a card goes straight to the document.
click-on File
click-on Start screen
wait 2
double-click-on Icon 16 x 16
wait 2
expect missing Create
expect canvas 16 16

# A recent entry can be taken off the list, from its context menu.
click-on File
click-on Start screen
wait 2
# (the document's own tab carries the same name, so the row is the second
# of two before, and the only one left after)
expect item start_fixture.lsprite @2
right-click-on start_fixture.lsprite @2
click-on Remove from this list
wait 2
expect missing start_fixture.lsprite @2
click-on Start with a blank canvas
