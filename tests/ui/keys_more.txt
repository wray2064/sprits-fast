# Commands with a key that no other script presses: the brush smaller, the layer
# stack, the next palette, a frame back, onion skin, the zoom out, snap, preview,
# the library, a paste onto a layer of its own, a flip, an inverted marquee, and
# the keys through the tabs.
key ctrl+0
tool pencil
colour e08f40

# Brush size, up and back down.
key shift+]
key shift+]
expect brush 3
key shift+[
expect brush 2
key shift+[
expect brush 1

# The stack: two layers; raised and lowered; merged down into one again (there
# has to be something drawn to merge).
click 3 3
click-on Add
expect layers 2
key ctrl+[
key ctrl+]
expect layers 2
key ctrl+e
expect layers 1
# Grouped, and ungrouped.
click-on Add
key ctrl+g
key ctrl+shift+g

# The next palette, by its key.
key ctrl+p
wait 2

# Frames: one more, then back one; onion skin on and off.
key t
click-on + Frame
expect frame 2
key ,
expect frame 1
expect flag onion off
key o
expect flag onion on
key o
expect flag onion off

# The zoom out by its key; snap to the tile grid, preview and the library.
zoom 8
key [
expect zoom 7
expect flag snap off
key shift+s
expect flag snap on
key shift+s
expect flag snap off
expect flag preview on
key p
expect flag preview off
key p
expect flag preview on
key ctrl+l
wait 2
expect flag library on
key ctrl+l
wait 2
expect flag library off
